   171→function formatDate(dateStr: string | null | undefined): string {
   172→  if (!dateStr) return '—'
   173→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   174→    day: '2-digit',
   175→    month: 'short',
   176→    year: 'numeric',
   177→  })
   178→}
   179→
   180→function formatDateTime(dateStr: string | null | undefined): string {
   181→  if (!dateStr) return '—'
   182→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   183→    day: '2-digit',
   184→    month: 'short',
   185→    year: 'numeric',
   186→    hour: '2-digit',
   187→    minute: '2-digit',
   188→  })
   189→}
   190→
   191→function isOverdue(dateEcheance: string | null | undefined, statut: string): boolean {
   192→  if (!dateEcheance || statut === 'terminee') return false
   193→  const today = new Date()
   194→  today.setHours(0, 0, 0, 0)
   195→  const dueDate = new Date(dateEcheance)
   196→  dueDate.setHours(0, 0, 0, 0)
   197→  return dueDate < today
   198→}

   204→function getPriorityConfig(priorite: string) {

   208→function getStatusConfig(statut: string) {

   212→function getInitials(nom: string): string {
   213→  return nom
   214→    .split(' ')
   215→    .map(w => w[0])
   216→    .join('')
   217→    .toUpperCase()
   218→    .slice(0, 2)
   219→}
