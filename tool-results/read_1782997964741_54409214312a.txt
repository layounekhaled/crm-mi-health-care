   117→// ─── Helper: normalize flags (Set or Array → Array) ──────────

   119→function normalizeFlags(flags: string[] | Set<string> | undefined | null): string[] {
   120→  if (!flags) return []
   121→  if (Array.isArray(flags)) return flags
   122→  if (flags instanceof Set) return [...flags]
   123→  return []
   124→}
   125→
   126→function hasFlag(flags: string[] | Set<string> | undefined | null, flag: string): boolean {
   127→  return normalizeFlags(flags).includes(flag)
   128→}

   163→function getFolderIcon(specialUse: string, path: string) {
   164→  if (specialUse === '\\Inbox' || path === 'INBOX') return <Inbox className="h-4 w-4" />
   165→  if (specialUse === '\\Sent') return <Send className="h-4 w-4" />
   166→  if (specialUse === '\\Drafts') return <FileText className="h-4 w-4" />
   167→  if (specialUse === '\\Trash') return <Trash2 className="h-4 w-4" />
   168→  if (specialUse === '\\Junk' || specialUse === '\\Spam') return <AlertCircle className="h-4 w-4" />
   169→  if (specialUse === '\\Flagged' || specialUse === '\\Starred') return <Star className="h-4 w-4" />
   170→  if (specialUse === '\\Archive') return <Archive className="h-4 w-4" />
   171→  // Fallback basé sur le chemin (support LWS/Dovecot français)
   172→  const lowerPath = path.toLowerCase()
   173→  if (lowerPath.includes('sent') || lowerPath.includes('envoy') || lowerPath.includes('éléments envoyés')) return <Send className="h-4 w-4" />
   174→  if (lowerPath.includes('draft') || lowerPath.includes('brouillon')) return <FileText className="h-4 w-4" />
   175→  if (lowerPath.includes('trash') || lowerPath.includes('corbeille') || lowerPath.includes('deleted') || lowerPath.includes('poubelle')) return <Trash2 className="h-4 w-4" />
   176→  if (lowerPath.includes('junk') || lowerPath.includes('spam') || lowerPath.includes('indésirable') || lowerPath.includes('courrier indésirable')) return <AlertCircle className="h-4 w-4" />
   177→  if (lowerPath.includes('flagged') || lowerPath.includes('starred') || lowerPath.includes('favori')) return <Star className="h-4 w-4" />
   178→  if (lowerPath.includes('archive') || lowerPath.includes('all mail') || lowerPath.includes('tous les messages')) return <Archive className="h-4 w-4" />
   179→  return <FolderOpen className="h-4 w-4" />
   180→}

   182→function getFolderLabel(path: string) {
   183→  if (path === 'INBOX') return 'Boîte de réception'
   184→  // Utiliser des correspondances partielles insensibles à la casse pour supporter LWS/Dovecot
   185→  const lowerPath = path.toLowerCase()
   186→  if (lowerPath.includes('sent') || lowerPath.includes('envoy') || lowerPath.includes('éléments envoyés')) return 'Envoyés'
   187→  if (lowerPath.includes('draft') || lowerPath.includes('brouillon')) return 'Brouillons'
   188→  if (lowerPath.includes('trash') || lowerPath.includes('corbeille') || lowerPath.includes('deleted') || lowerPath.includes('poubelle')) return 'Corbeille'
   189→  if (lowerPath.includes('junk') || lowerPath.includes('spam') || lowerPath.includes('indésirable') || lowerPath.includes('courrier indésirable')) return 'Indésirables'
   190→  if (lowerPath.includes('flagged') || lowerPath.includes('starred') || lowerPath.includes('favori')) return 'Favoris'
   191→  if (lowerPath.includes('archive') || lowerPath.includes('all mail') || lowerPath.includes('tous les messages')) return 'Archives'
   192→  // Pour les dossiers Gmail qui commencent par [Gmail]/, afficher un nom plus propre
   193→  if (path.startsWith('[Gmail]/')) {
   194→    const gmailName = path.replace('[Gmail]/', '')
   195→    return gmailName.replace(/([A-Z])/g, ' $1').trim()
   196→  }
   197→  // Pour les dossiers INBOX.xxx (Dovecot), afficher le nom après le séparateur
   198→  if (path.startsWith('INBOX.')) {
   199→    return path.replace('INBOX.', '')
   200→  }
   201→  return path
   202→}

   204→function formatDate(dateStr: string) {
   205→  try {
   206→    const d = new Date(dateStr)
   207→    const now = new Date()
   208→    const diffMs = now.getTime() - d.getTime()
   209→    const diffDays = Math.floor(diffMs / 86400000)
   210→    if (diffDays < 1) return d.toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })
   211→    if (diffDays < 7) return d.toLocaleDateString('fr-FR', { weekday: 'short' })
   212→    return d.toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' })
   213→  } catch {
   214→    return ''
   215→  }
   216→}

   229→function extractSenderName(from: { name: string; address: string }[]) {
   230→  if (!from || from.length === 0) return 'Inconnu'
   231→  const sender = from[0]
   232→  if (sender.name) return sender.name
   233→  return sender.address?.split('@')[0] || 'Inconnu'
   234→}

   236→function extractEmailAddress(from: { name: string; address: string }[]) {
   237→  if (!from || from.length === 0) return ''
   238→  return from.map((f) => f.name ? `${f.name} <${f.address}>` : f.address).join(', ')
   239→}

   247→function hasAttachments(bodyStructure?: BodyStructurePart): boolean {
   248→  if (!bodyStructure) return false
   249→  const check = (part: BodyStructurePart): boolean => {
   250→    if (part.disposition === 'attachment') return true
   251→    if (part.childNodes) return part.childNodes.some(check)
   252→    return false
   253→  }
   254→  return check(bodyStructure)
   255→}

   257→function sanitizeHtml(html: string): string {
   258→  return DOMPurify.sanitize(html, {
   259→    ALLOWED_TAGS: [
   260→      'h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'p', 'br', 'hr', 'div', 'span',
   261→      'table', 'thead', 'tbody', 'tfoot', 'tr', 'th', 'td', 'caption',
   262→      'ul', 'ol', 'li', 'dl', 'dt', 'dd',
   263→      'a', 'img', 'figure', 'figcaption',
   264→      'b', 'i', 'u', 's', 'em', 'strong', 'small', 'sub', 'sup',
   265→      'blockquote', 'pre', 'code',
   266→      'font', 'center',
   267→    ],
   268→    ALLOWED_ATTR: [
   269→      'href', 'src', 'alt', 'title', 'width', 'height', 'style',
   270→      'class', 'id', 'colspan', 'rowspan', 'cellpadding', 'cellspacing',
   271→      'border', 'align', 'valign', 'bgcolor', 'color', 'size', 'face',
   272→      'target', 'rel',
   273→    ],
   274→    ALLOW_DATA_ATTR: false,
   275→    ADD_ATTR: ['target'],
   276→  })
   277→}
