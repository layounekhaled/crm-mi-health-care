   117→// ─── Helper: normalize flags (Set or Array → Array) ──────────

   119→function normalizeFlags(flags: string[] | Set<string> | undefined | null): string[] {
   120→  if (!flags) return []
   121→  if (Array.isArray(flags)) return flags
   122→  if (flags instanceof Set) return [...flags]
   123→  return []
   124→}
   125→
   126→function hasFlag(flags: string[] | Set<string> | undefined | null, flag: string): boolean {
   127→  return normalizeFlags(flags).includes(flag)
   128→}

   177→function getFolderLabel(path: string) {
   178→  if (path === 'INBOX') return 'Boîte de réception'
   179→  // Utiliser des correspondances partielles insensibles à la casse pour supporter LWS/Dovecot
   180→  const lowerPath = path.toLowerCase()
   181→  if (lowerPath.includes('sent') || lowerPath.includes('envoy') || lowerPath.includes('éléments envoyés')) return 'Envoyés'
   182→  if (lowerPath.includes('draft') || lowerPath.includes('brouillon')) return 'Brouillons'
   183→  if (lowerPath.includes('trash') || lowerPath.includes('corbeille') || lowerPath.includes('deleted') || lowerPath.includes('poubelle')) return 'Corbeille'
   184→  if (lowerPath.includes('junk') || lowerPath.includes('spam') || lowerPath.includes('indésirable') || lowerPath.includes('courrier indésirable')) return 'Indésirables'
   185→  if (lowerPath.includes('flagged') || lowerPath.includes('starred') || lowerPath.includes('favori')) return 'Favoris'
   186→  if (lowerPath.includes('archive') || lowerPath.includes('all mail') || lowerPath.includes('tous les messages')) return 'Archives'
   187→  // Pour les dossiers Gmail qui commencent par [Gmail]/, afficher un nom plus propre
   188→  if (path.startsWith('[Gmail]/')) {
   189→    const gmailName = path.replace('[Gmail]/', '')
   190→    return gmailName.replace(/([A-Z])/g, ' $1').trim()
   191→  }
   192→  // Pour les dossiers INBOX.xxx (Dovecot), afficher le nom après le séparateur
   193→  if (path.startsWith('INBOX.')) {
   194→    return path.replace('INBOX.', '')
   195→  }
   196→  return path
   197→}

   199→function formatDate(dateStr: string) {
   200→  try {
   201→    const d = new Date(dateStr)
   202→    const now = new Date()
   203→    const diffMs = now.getTime() - d.getTime()
   204→    const diffDays = Math.floor(diffMs / 86400000)
   205→    if (diffDays < 1) return d.toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })
   206→    if (diffDays < 7) return d.toLocaleDateString('fr-FR', { weekday: 'short' })
   207→    return d.toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' })
   208→  } catch {
   209→    return ''
   210→  }
   211→}

   224→function extractSenderName(from: { name: string; address: string }[]) {
   225→  if (!from || from.length === 0) return 'Inconnu'
   226→  const sender = from[0]
   227→  if (sender.name) return sender.name
   228→  return sender.address?.split('@')[0] || 'Inconnu'
   229→}

   231→function extractEmailAddress(from: { name: string; address: string }[]) {
   232→  if (!from || from.length === 0) return ''
   233→  return from.map((f) => f.name ? `${f.name} <${f.address}>` : f.address).join(', ')
   234→}

   242→function hasAttachments(bodyStructure?: BodyStructurePart): boolean {
   243→  if (!bodyStructure) return false
   244→  const check = (part: BodyStructurePart): boolean => {
   245→    if (part.disposition === 'attachment') return true
   246→    if (part.childNodes) return part.childNodes.some(check)
   247→    return false
   248→  }
   249→  return check(bodyStructure)
   250→}

   252→function sanitizeHtml(html: string): string {
   253→  return DOMPurify.sanitize(html, {
   254→    ALLOWED_TAGS: [
   255→      'h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'p', 'br', 'hr', 'div', 'span',
   256→      'table', 'thead', 'tbody', 'tfoot', 'tr', 'th', 'td', 'caption',
   257→      'ul', 'ol', 'li', 'dl', 'dt', 'dd',
   258→      'a', 'img', 'figure', 'figcaption',
   259→      'b', 'i', 'u', 's', 'em', 'strong', 'small', 'sub', 'sup',
   260→      'blockquote', 'pre', 'code',
   261→      'font', 'center',
   262→    ],
   263→    ALLOWED_ATTR: [
   264→      'href', 'src', 'alt', 'title', 'width', 'height', 'style',
   265→      'class', 'id', 'colspan', 'rowspan', 'cellpadding', 'cellspacing',
   266→      'border', 'align', 'valign', 'bgcolor', 'color', 'size', 'face',
   267→      'target', 'rel',
   268→    ],
   269→    ALLOW_DATA_ATTR: false,
   270→    ADD_ATTR: ['target'],
   271→  })
   272→}
