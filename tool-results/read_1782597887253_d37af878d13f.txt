    88→function formatTime(dateStr: string) {
    89→  const d = new Date(dateStr)
    90→  const now = new Date()
    91→  const diffMs = now.getTime() - d.getTime()
    92→  const diffMins = Math.floor(diffMs / 60000)
    93→  const diffHours = Math.floor(diffMs / 3600000)
    94→  const diffDays = Math.floor(diffMs / 86400000)
    95→
    96→  if (diffMins < 1) return "à l'instant"
    97→  if (diffMins < 60) return `il y a ${diffMins}min`
    98→  if (diffHours < 24) return `il y a ${diffHours}h`
    99→  if (diffDays < 7) return `il y a ${diffDays}j`
   100→  return d.toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' })
   101→}

   104→function playNotificationSound() {
   105→  try {
   106→    const audioContext = new (window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext)()
   107→    const oscillator = audioContext.createOscillator()
   108→    const gainNode = audioContext.createGain()
   109→    oscillator.connect(gainNode)
   110→    gainNode.connect(audioContext.destination)
   111→    oscillator.frequency.setValueAtTime(800, audioContext.currentTime)
   112→    oscillator.frequency.setValueAtTime(1000, audioContext.currentTime + 0.1)
   113→    oscillator.frequency.setValueAtTime(800, audioContext.currentTime + 0.2)
   114→    gainNode.gain.setValueAtTime(0.15, audioContext.currentTime)
   115→    gainNode.gain.exponentialRampToValueAtTime(0.01, audioContext.currentTime + 0.35)
   116→    oscillator.start(audioContext.currentTime)
   117→    oscillator.stop(audioContext.currentTime + 0.35)
   118→  } catch {
   119→    // Audio non supporté, on ignore
   120→  }
   121→}
