   101→
   102→export default function DocumentsModule() {
   103→  const { user, hasPermission } = useAuth()
   104→  const canCreate = hasPermission('documents', 'create')
   105→  const canEdit = hasPermission('documents', 'edit')
   106→  const canDelete = hasPermission('documents', 'delete')
   107→  const isAdmin = user?.role === 'admin'
   108→
   109→  // State
   110→  const [documents, setDocuments] = useState<DocumentRecord[]>([])
   111→  const [sends, setSends] = useState<DocumentSendRecord[]>([])
   112→  const [prospects, setProspects] = useState<ProspectOption[]>([])
   113→  const [loading, setLoading] = useState(true)
   114→  const [searchQuery, setSearchQuery] = useState('')
   115→  const [filterBrand, setFilterBrand] = useState<string>('all')
   116→  const [filterType, setFilterType] = useState<string>('all')
   117→  const [activeTab, setActiveTab] = useState<string>('library')
   118→  const [showArchived, setShowArchived] = useState(false)
   119→
   120→  // Upload dialog
   121→  const [showUpload, setShowUpload] = useState(false)
   122→  const [uploadForm, setUploadForm] = useState({
   123→    title: '', description: '', brand: 'MIR', productName: '', documentType: 'catalogue'
   124→  })
   125→  const [uploadFile, setUploadFile] = useState<File | null>(null)
   126→  const [uploading, setUploading] = useState(false)
   127→
   128→  // Edit dialog
   129→  const [showEdit, setShowEdit] = useState(false)
   130→  const [editingDoc, setEditingDoc] = useState<DocumentRecord | null>(null)
   131→  const [editForm, setEditForm] = useState({
   132→    title: '', description: '', brand: '', productName: '', documentType: '', status: ''
   133→  })
   134→
   135→  // Send dialog
   136→  const [showSend, setShowSend] = useState(false)
   137→  const [sendForm, setSendForm] = useState({
   138→    recipientType: 'manual', recipientId: '', recipientEmail: '', message: ''
   139→  })
   140→  const [sending, setSending] = useState(false)
   141→
   142→  // Selection
   143→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   144→
   145→  // Delete confirm
   146→  const [showDeleteConfirm, setShowDeleteConfirm] = useState(false)
   147→  const [deleteTarget, setDeleteTarget] = useState<DocumentRecord | null>(null)
   148→
   149→  // Fetch documents
   150→  const fetchDocuments = useCallback(async () => {
   151→    try {
   152→      const params = new URLSearchParams()
   153→      if (filterBrand !== 'all') params.set('brand', filterBrand)
   154→      if (filterType !== 'all') params.set('documentType', filterType)
   155→      if (searchQuery) params.set('search', searchQuery)
   156→      if (showArchived) params.set('status', 'archived')
   157→      params.set('limit', '100')
   158→
   159→      const res = await fetch(`/api/documents?${params}`)
   160→      const data = await res.json()
   161→      if (data.data) setDocuments(data.data)
   162→    } catch (error) {
   163→      console.error('Fetch documents error:', error)
   164→    }
   165→  }, [filterBrand, filterType, searchQuery, showArchived])
   166→
   167→  // Fetch sends history
   168→  const fetchSends = useCallback(async () => {
   169→    try {
   170→      const res = await fetch('/api/documents/sends?limit=50')
   171→      const data = await res.json()
   172→      if (data.data) setSends(data.data)
   173→    } catch (error) {
   174→      console.error('Fetch sends error:', error)
   175→    }
   176→  }, [])
   177→
   178→  // Fetch prospects for send dialog
   179→  const fetchProspects = useCallback(async () => {
   180→    try {
   181→      const res = await fetch('/api/prospects?limit=500')
   182→      const data = await res.json()
   183→      if (data.prospects) {
   184→        setProspects(data.prospects.map((p: any) => ({
   185→          id: p.id, nom: p.nom, email: p.email, telephone: p.telephone
   186→        })))
   187→      }
   188→    } catch {}
   189→  }, [])
   190→
   191→  // Initial load
   192→  useEffect(() => {
   193→    const load = async () => {
   194→      setLoading(true)
   195→      await Promise.all([fetchDocuments(), fetchSends(), fetchProspects()])
   196→      setLoading(false)
   197→    }
   198→    load()
   199→  }, [fetchDocuments, fetchSends, fetchProspects])
   200→
   201→  // Refetch when filters change
   202→  useEffect(() => {
   203→    fetchDocuments()
   204→  }, [fetchDocuments])
   205→
   206→  // Upload handler
   207→  const handleUpload = async () => {
   208→    if (!uploadFile || !uploadForm.title) {
   209→      toast.error('Fichier et titre sont requis')
   210→      return
   211→    }
   212→
   213→    setUploading(true)
   214→    try {
   215→      const formData = new FormData()
   216→      formData.append('file', uploadFile)
   217→      formData.append('title', uploadForm.title)
   218→      formData.append('description', uploadForm.description)
   219→      formData.append('brand', uploadForm.brand)
   220→      formData.append('productName', uploadForm.productName)
   221→      formData.append('documentType', uploadForm.documentType)
   222→
   223→      const res = await fetch('/api/documents', { method: 'POST', body: formData })
   224→      const data = await res.json()
   225→
   226→      if (!res.ok) {
   227→        toast.error(data.error || 'Erreur lors de l\'upload')
   228→        return
   229→      }
   230→
   231→      toast.success('Document ajouté avec succès')
   232→      setShowUpload(false)
   233→      setUploadForm({ title: '', description: '', brand: 'MIR', productName: '', documentType: 'catalogue' })
   234→      setUploadFile(null)
   235→      fetchDocuments()
   236→    } catch (error: any) {
   237→      console.error('Upload error:', error)
   238→      toast.error(error?.message || 'Erreur lors de l\'upload — vérifiez la connexion réseau')
   239→    } finally {
   240→      setUploading(false)
   241→    }
   242→  }
   243→
   244→  // Edit handler
   245→  const handleEdit = async () => {
   246→    if (!editingDoc) return
   247→
   248→    try {
   249→      const res = await fetch(`/api/documents/${editingDoc.id}`, {
   250→        method: 'PUT',
   251→        headers: { 'Content-Type': 'application/json' },
   252→        body: JSON.stringify(editForm),
   253→      })
   254→      const data = await res.json()
   255→
   256→      if (!res.ok) {
   257→        toast.error(data.error || 'Erreur lors de la modification')
   258→        return
   259→      }
   260→
   261→      toast.success('Document modifié')
   262→      setShowEdit(false)
   263→      setEditingDoc(null)
   264→      fetchDocuments()
   265→    } catch {
   266→      toast.error('Erreur lors de la modification')
   267→    }
   268→  }
   269→
   270→  // Delete handler
   271→  const handleDelete = async () => {
   272→    if (!deleteTarget) return
   273→
   274→    try {
   275→      const res = await fetch(`/api/documents/${deleteTarget.id}`, { method: 'DELETE' })
   276→      if (!res.ok) {
   277→        const data = await res.json()
   278→        toast.error(data.error || 'Erreur lors de la suppression')
   279→        return
   280→      }
   281→
   282→      toast.success('Document supprimé')
   283→      setShowDeleteConfirm(false)
   284→      setDeleteTarget(null)
   285→      fetchDocuments()
   286→    } catch {
   287→      toast.error('Erreur lors de la suppression')
   288→    }
   289→  }
   290→
   291→  // Archive/Restore handler
   292→  const handleArchive = async (doc: DocumentRecord, newStatus: string) => {
   293→    try {
   294→      const res = await fetch(`/api/documents/${doc.id}`, {
   295→        method: 'PUT',
   296→        headers: { 'Content-Type': 'application/json' },
   297→        body: JSON.stringify({ status: newStatus }),
   298→      })
   299→
   300→      if (!res.ok) {
   301→        toast.error('Erreur')
   302→        return
   303→      }
   304→
   305→      toast.success(newStatus === 'archived' ? 'Document archivé' : 'Document restauré')
   306→      fetchDocuments()
   307→    } catch {
   308→      toast.error('Erreur')
   309→    }
   310→  }
   311→
   312→  // Send documents handler
   313→  const handleSend = async () => {
   314→    if (!sendForm.recipientEmail && sendForm.recipientType === 'manual') {
   315→      toast.error('Email du destinataire requis')
   316→      return
   317→    }
   318→
   319→    if (selectedIds.size === 0) {
   320→      toast.error('Sélectionnez au moins un document')
   321→      return
   322→    }
   323→
   324→    setSending(true)
   325→    try {
   326→      const res = await fetch('/api/documents/send', {
   327→        method: 'POST',
   328→        headers: { 'Content-Type': 'application/json' },
   329→        body: JSON.stringify({
   330→          documentIds: Array.from(selectedIds),
   331→          recipientType: sendForm.recipientType,
   332→          recipientId: sendForm.recipientId || null,
   333→          recipientEmail: sendForm.recipientEmail,
   334→          message: sendForm.message || null,
   335→        }),
   336→      })
   337→
   338→      const data = await res.json()
   339→
   340→      if (!res.ok) {
   341→        toast.error(data.error || 'Erreur lors de l\'envoi')
   342→        return
   343→      }
   344→
   345→      toast.success(`${selectedIds.size} document(s) envoyé(s) avec succès`)
   346→      setShowSend(false)
   347→      setSendForm({ recipientType: 'manual', recipientId: '', recipientEmail: '', message: '' })
   348→      setSelectedIds(new Set())
   349→      fetchSends()
   350→    } catch {
   351→      toast.error('Erreur lors de l\'envoi')
   352→    } finally {
   353→      setSending(false)
   354→    }
   355→  }
   356→
   357→  // Copy link
   358→  const copyLink = (url: string) => {
   359→    navigator.clipboard.writeText(url)
   360→    toast.success('Lien copié')
   361→  }
   362→
   363→  // Toggle selection
   364→  const toggleSelect = (id: string) => {
   365→    const next = new Set(selectedIds)
   366→    if (next.has(id)) next.delete(id)
   367→    else next.add(id)
   368→    setSelectedIds(next)
   369→  }
   370→
   371→  // Select all visible
   372→  const selectAll = () => {
   373→    if (selectedIds.size === documents.length) {
   374→      setSelectedIds(new Set())
   375→    } else {
   376→      setSelectedIds(new Set(documents.map(d => d.id)))
   377→    }
   378→  }
   379→
   380→  // Open edit dialog
   381→  const openEdit = (doc: DocumentRecord) => {
   382→    setEditingDoc(doc)
   383→    setEditForm({
   384→      title: doc.title,
   385→      description: doc.description || '',
   386→      brand: doc.brand,
   387→      productName: doc.productName || '',
   388→      documentType: doc.documentType,
   389→      status: doc.status,
   390→    })
   391→    setShowEdit(true)
   392→  }
   393→
   394→  // Open send dialog
   395→  const openSend = (docIds?: string[]) => {
   396→    if (docIds && docIds.length > 0) {
   397→      setSelectedIds(new Set(docIds))
   398→    }
   399→    if (selectedIds.size === 0 && !docIds) {
   400→      toast.error('Sélectionnez au moins un document')
   401→      return
   402→    }
   403→    setShowSend(true)
   404→  }
   405→
   406→  // Handle prospect selection for send
   407→  const handleProspectSelect = (prospectId: string) => {
   408→    const prospect = prospects.find(p => p.id === prospectId)
   409→    setSendForm(prev => ({
   410→      ...prev,
   411→      recipientId: prospectId,
   412→      recipientEmail: prospect?.email || prospect?.telephone || '',
   413→    }))
   414→  }
   415→
   416→  const getTypeLabel = (type: string) => DOCUMENT_TYPES.find(t => t.value === type)?.label || type
   417→
   418→  return (
   419→    <div className="min-h-screen bg-slate-50">
   420→      <ModuleHeader
   421→        title="Documents"
   422→        subtitle="Bibliothèque de documents PDF — MI HEALTH CARE"
   423→        actions={
   424→          <div className="flex items-center gap-2">
   425→            {isAdmin && (
   426→              <Button
   427→                onClick={async () => {
   428→                  try {
   429→                    const res = await fetch('/api/documents/init-bucket', { method: 'POST' })
   430→                    const data = await res.json()
   431→                    if (res.ok) toast.success(data.message)
   432→                    else toast.error(data.error)
   433→                  } catch {
   434→                    toast.error('Erreur initialisation bucket')
   435→                  }
   436→                }}
   437→                variant="outline"
   438→                size="sm"
   439→                className="border-[#134885] text-[#134885] hover:bg-[#134885] hover:text-white"
   440→              >
   441→                <FolderOpen className="mr-1.5 h-4 w-4" />
   442→                Init Stockage
   443→              </Button>
   444→            )}
   445→            {canCreate && (
   446→              <Button
   447→                onClick={() => setShowUpload(true)}
   448→                className="bg-[#F6852A] hover:bg-[#e5761f] text-white"
   449→                size="sm"
   450→              >
   451→                <Upload className="mr-1.5 h-4 w-4" />
   452→                Ajouter
   453→              </Button>
   454→            )}
   455→            {selectedIds.size > 0 && (
   456→              <Button
   457→                onClick={() => openSend()}
   458→                className="bg-[#134885] hover:bg-[#0D3A6E] text-white"
   459→                size="sm"
   460→              >
   461→                <Send className="mr-1.5 h-4 w-4" />
   462→                Envoyer ({selectedIds.size})
   463→              </Button>
   464→            )}
   465→          </div>
   466→        }
   467→      />
   468→
   469→      <div className="mx-auto max-w-7xl px-4 py-4 sm:px-6 lg:px-8">
   470→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   471→          <TabsList className="mb-4 bg-white shadow-sm">
   472→            <TabsTrigger value="library" className="gap-1.5">
   473→              <BookOpen className="h-4 w-4" />
   474→              Bibliothèque
   475→            </TabsTrigger>
   476→            <TabsTrigger value="history" className="gap-1.5">
   477→              <History className="h-4 w-4" />
   478→              Historique
   479→            </TabsTrigger>
   480→          </TabsList>
   481→
   482→          {/* ====== LIBRARY TAB ====== */}
   483→          <TabsContent value="library">
   484→            {/* Filters bar */}
   485→            <div className="mb-4 flex flex-col sm:flex-row gap-3">
   486→              <div className="relative flex-1">
   487→                <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
   488→                <Input
   489→                  placeholder="Rechercher un document..."
   490→                  value={searchQuery}
   491→                  onChange={(e) => setSearchQuery(e.target.value)}
   492→                  className="pl-9"
   493→                />
   494→              </div>
   495→              <Select value={filterBrand} onValueChange={setFilterBrand}>
   496→                <SelectTrigger className="w-[160px]">
   497→                  <SelectValue placeholder="Marque" />
   498→                </SelectTrigger>
   499→                <SelectContent>
   500→                  <SelectItem value="all">Toutes les marques</SelectItem>
   501→                  {BRANDS.map(b => (
   502→                    <SelectItem key={b} value={b}>{b}</SelectItem>
   503→                  ))}
   504→                </SelectContent>
   505→              </Select>
   506→              <Select value={filterType} onValueChange={setFilterType}>
   507→                <SelectTrigger className="w-[160px]">
   508→                  <SelectValue placeholder="Type" />
   509→                </SelectTrigger>
   510→                <SelectContent>
   511→                  <SelectItem value="all">Tous les types</SelectItem>
   512→                  {DOCUMENT_TYPES.map(t => (
   513→                    <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>
   514→                  ))}
   515→                </SelectContent>
   516→              </Select>
   517→              {canEdit && (
   518→                <Button
   519→                  variant={showArchived ? 'default' : 'outline'}
   520→                  size="sm"
   521→                  onClick={() => setShowArchived(!showArchived)}
   522→                  className={showArchived ? 'bg-slate-600' : ''}
   523→                >
   524→                  <Archive className="mr-1.5 h-4 w-4" />
   525→                  {showArchived ? 'Voir actifs' : 'Archivés'}
   526→                </Button>
   527→              )}
   528→            </div>
   529→
   530→            {/* Selection bar */}
   531→            {documents.length > 0 && (
   532→              <div className="mb-3 flex items-center gap-3">
   533→                <div className="flex items-center gap-2">
   534→                  <Checkbox
   535→                    checked={selectedIds.size === documents.length && documents.length > 0}
   536→                    onCheckedChange={selectAll}
   537→                  />
   538→                  <span className="text-sm text-slate-500">
   539→                    {selectedIds.size > 0 ? `${selectedIds.size} sélectionné(s)` : 'Tout sélectionner'}
   540→                  </span>
   541→                </div>
   542→                {selectedIds.size > 0 && (
   543→                  <Button
   544→                    variant="ghost"
   545→                    size="sm"
   546→                    onClick={() => setSelectedIds(new Set())}
   547→                    className="text-slate-500"
   548→                  >
   549→                    <X className="mr-1 h-3 w-3" />
   550→                    Désélectionner
   551→                  </Button>
   552→                )}
   553→              </div>
   554→            )}
   555→
   556→            {/* Documents grid */}
   557→            {loading ? (
   558→              <div className="flex items-center justify-center py-20">
   559→                <Loader2 className="h-8 w-8 animate-spin text-[#134885]" />
   560→              </div>
   561→            ) : documents.length === 0 ? (
   562→              <div className="flex flex-col items-center justify-center py-20 text-slate-400">
   563→                <FileText className="h-16 w-16 mb-4 opacity-30" />
   564→                <p className="text-lg font-medium">Aucun document trouvé</p>
   565→                <p className="text-sm mt-1">
   566→                  {canCreate ? 'Cliquez sur "Ajouter" pour uploader votre premier document' : 'Aucun document actif disponible'}
   567→                </p>
   568→              </div>
   569→            ) : (
   570→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   571→                <AnimatePresence>
   572→                  {documents.map((doc) => (
   573→                    <motion.div
   574→                      key={doc.id}
   575→                      initial={{ opacity: 0, y: 10 }}
   576→                      animate={{ opacity: 1, y: 0 }}
   577→                      exit={{ opacity: 0, y: -10 }}
