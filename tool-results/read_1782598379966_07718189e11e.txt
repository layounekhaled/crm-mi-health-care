   107→function getAvatarColor(nom: string) {
   108→  let hash = 0
   109→  for (let i = 0; i < nom.length; i++) {
   110→    hash = nom.charCodeAt(i) + ((hash << 5) - hash)
   111→  }
   112→  return avatarColors[Math.abs(hash) % avatarColors.length]
   113→}

   127→function formatTime(dateStr: string) {
   128→  const d = new Date(dateStr)
   129→  const now = new Date()
   130→  const diffMs = now.getTime() - d.getTime()
   131→  const diffMins = Math.floor(diffMs / 60000)
   132→  const diffHours = Math.floor(diffMs / 3600000)
   133→  const diffDays = Math.floor(diffMs / 86400000)
   134→
   135→  if (diffMins < 1) return "à l'instant"
   136→  if (diffMins < 60) return `il y a ${diffMins}min`
   137→  if (diffHours < 24) return d.toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })
   138→  if (diffDays < 7) return d.toLocaleDateString('fr-FR', { weekday: 'short', hour: '2-digit', minute: '2-digit' })
   139→  return d.toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' })
   140→}

   147→// Vérifier si deux dates sont le même jour
   148→function isSameDay(d1: Date, d2: Date) {
   149→  return d1.getFullYear() === d2.getFullYear() &&
   150→    d1.getMonth() === d2.getMonth() &&
   151→    d1.getDate() === d2.getDate()
   152→}

   154→function formatDateSeparator(dateStr: string) {
   155→  const d = new Date(dateStr)
   156→  const now = new Date()
   157→  const diffDays = Math.floor((now.getTime() - d.getTime()) / 86400000)
   158→
   159→  if (isSameDay(d, now)) return "Aujourd'hui"
   160→  const yesterday = new Date(now)
   161→  yesterday.setDate(yesterday.getDate() - 1)
   162→  if (isSameDay(d, yesterday)) return 'Hier'
   163→
   164→  return d.toLocaleDateString('fr-FR', { weekday: 'long', day: 'numeric', month: 'long' })
   165→}

   167→// Vérifier si un texte contient une URL
   168→function linkifyText(text: string) {
   169→  const urlRegex = /(https?:\/\/[^\s<]+)/g
   170→  const parts = text.split(urlRegex)
   171→  return parts.map((part, i) => {
   172→    if (urlRegex.test(part)) {
   173→      return (
   174→        <a
   175→          key={i}
   176→          href={part}
   177→          target="_blank"
   178→          rel="noopener noreferrer"
   179→          className="underline underline-offset-2 hover:opacity-80 break-all"
   180→          onClick={(e) => e.stopPropagation()}
   181→        >
   182→          {part.length > 50 ? part.slice(0, 50) + '...' : part}
   183→        </a>
   184→      )
   185→    }
   186→    return <span key={i}>{part}</span>
   187→  })
   188→}

   190→// Son de notification
   191→function playNotificationSound() {
   192→  try {
   193→    const audioContext = new (window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext)()
   194→    const oscillator = audioContext.createOscillator()
   195→    const gainNode = audioContext.createGain()
   196→    oscillator.connect(gainNode)
   197→    gainNode.connect(audioContext.destination)
   198→    oscillator.frequency.setValueAtTime(800, audioContext.currentTime)
   199→    oscillator.frequency.setValueAtTime(1000, audioContext.currentTime + 0.1)
   200→    oscillator.frequency.setValueAtTime(800, audioContext.currentTime + 0.2)
   201→    gainNode.gain.setValueAtTime(0.15, audioContext.currentTime)
   202→    gainNode.gain.exponentialRampToValueAtTime(0.01, audioContext.currentTime + 0.35)
   203→    oscillator.start(audioContext.currentTime)
   204→    oscillator.stop(audioContext.currentTime + 0.35)
   205→  } catch {
   206→    // Audio non supporté
   207→  }
   208→}
